%% Wireless Communication Performance Analysis
% This script analyzes WiFi, Bluetooth, and Cellular communication
% performance using the Friis Transmission Equation.
% It calculates:
% 1. Received Power (dB)
% 2. SINR (dB)
% 3. Channel Capacity

clear all;
close all;
clc;

%% General Parameters
G_t = 1;              % Transmitter antenna gain
G_r = 1;              % Receiver antenna gain
c = 3*10^8;           % Speed of light (m/s)
Pn = 500*1e-6;        % Noise power (W)

%% ===================== WIFI SIMULATION =====================
f_wifi = 2.4*1e9;          % WiFi frequency (Hz)
BW_wifi = 22*1e6;          % Bandwidth (Hz)
d_wifi = 1:5:100;          % Distance range (m)
P_t_wifi = 100*1e-3;       % Transmit power (W)
lamb_wifi = c/f_wifi;      % Wavelength

% Received Power using Friis equation
Pr_wifi = (P_t_wifi*G_t*G_r*(lamb_wifi^2))./((4*pi*d_wifi).^2);

% Convert received power to dB
Pr_wifi_dB = 10*log10(Pr_wifi);

% SINR calculation
SINR_wifi = Pr_wifi./Pn;
SINR_wifi_dB = 10*log10(SINR_wifi);

% Channel Capacity
C_wifi = BW_wifi .* log2(1 + SINR_wifi);

%% ===================== BLUETOOTH SIMULATION =====================
f_bluetooth = 2.45*1e9;     % Frequency (Hz)
BW_bluetooth = 2*1e6;       % Bandwidth
d_bluetooth = 0.5:0.5:10;   % Distance
P_t_bluetooth = 10*1e-3;    % Transmit power
lamb_bluetooth = c/f_bluetooth;

Pr_bt = (P_t_bluetooth*G_t*G_r*(lamb_bluetooth^2))./((4*pi*d_bluetooth).^2);
Pr_bt_dB = 10*log10(Pr_bt);

SINR_bt = Pr_bt./Pn;
SINR_bt_dB = 10*log10(SINR_bt);

C_bt = BW_bluetooth .* log2(1 + SINR_bt);

%% ===================== CELLULAR SIMULATION =====================
f_cellular = 850*1e6;       % Frequency (Hz)
BW_cellular = 200*1e3;      % Bandwidth
d_cellular = 100:100:5000;  % Distance
P_t_cellular = 40;          % Transmit power
lamb_cellular = c/f_cellular;

Pr_cell = (P_t_cellular*G_t*G_r*(lamb_cellular^2))./((4*pi*d_cellular).^2);
Pr_cell_dB = 10*log10(Pr_cell);

SINR_cell = Pr_cell./Pn;
SINR_cell_dB = 10*log10(SINR_cell);

C_cell = BW_cellular .* log2(1 + SINR_cell);

%% ===================== PLOTTING RESULTS =====================

figure;

% -------- WIFI --------
subplot(3,3,1)
plot(d_wifi, Pr_wifi_dB,'LineWidth',1.5)
title('WiFi')
ylabel('P_r (dB)')
xlabel('distance (m)')
grid on

subplot(3,3,4)
plot(d_wifi, SINR_wifi_dB,'LineWidth',1.5)
ylabel('SINR (dB)')
xlabel('distance (m)')
grid on

subplot(3,3,7)
plot(d_wifi, C_wifi/1e6,'LineWidth',1.5)
ylabel('Capacity (Mbps)')
xlabel('distance (m)')
grid on

% -------- BLUETOOTH --------
subplot(3,3,2)
plot(d_bluetooth, Pr_bt_dB,'LineWidth',1.5)
title('Bluetooth')
xlabel('distance (m)')
grid on

subplot(3,3,5)
plot(d_bluetooth, SINR_bt_dB,'LineWidth',1.5)
xlabel('distance (m)')
grid on

subplot(3,3,8)
plot(d_bluetooth, C_bt/1e6,'LineWidth',1.5)
xlabel('distance (m)')
grid on

% -------- CELLULAR --------
subplot(3,3,3)
plot(d_cellular, Pr_cell_dB,'LineWidth',1.5)
title('Cellular')
xlabel('distance (m)')
grid on

subplot(3,3,6)
plot(d_cellular, SINR_cell_dB,'LineWidth',1.5)
xlabel('distance (m)')
grid on

subplot(3,3,9)
plot(d_cellular, C_cell/1e6,'LineWidth',1.5)
xlabel('distance (m)')
grid on

sgtitle('Wireless Communication Performance Comparison')

%% ===================== COMPARATIVE ANALYSIS =====================

figure

plot(d_wifi, C_wifi/1e6,'LineWidth',2)
hold on
plot(d_bluetooth, C_bt/1e6,'LineWidth',2)
plot(d_cellular, C_cell/1e6,'LineWidth',2)

legend('WiFi','Bluetooth','Cellular')
xlabel('Distance (m)')
ylabel('Capacity (Mbps)')
title('Capacity Comparison')
grid on
